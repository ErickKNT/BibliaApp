  (function initDrawerInteractions() {

    // Privacy Toggle Behavior
    const btnPrivate = document.getElementById('privacy-toggle-private');
    const btnShare = document.getElementById('privacy-toggle-share');
    let isPrivate = true;

    if (btnPrivate && btnShare) {
      btnPrivate.addEventListener('click', () => {
        isPrivate = true;
        btnPrivate.className = 'px-2.5 py-1 rounded-full font-label-sm text-label-sm transition-all bg-surface-container-highest text-primary font-medium flex items-center gap-1';
        btnShare.className = 'px-2.5 py-1 rounded-full font-label-sm text-label-sm transition-all text-on-surface-variant hover:text-primary flex items-center gap-1';
      });

      btnShare.addEventListener('click', () => {
        isPrivate = false;
        btnShare.className = 'px-2.5 py-1 rounded-full font-label-sm text-label-sm transition-all bg-surface-container-highest text-primary font-medium flex items-center gap-1';
        btnPrivate.className = 'px-2.5 py-1 rounded-full font-label-sm text-label-sm transition-all text-on-surface-variant hover:text-primary flex items-center gap-1';
      });
    }

    // Like toggle micro-interaction
    const likeBtn = document.querySelector('.like-btn');
    if (likeBtn) {
      likeBtn.addEventListener('click', function() {
        const icon = this.querySelector('.material-symbols-outlined');
        const counter = this.querySelector('.like-counter');
        const currentCount = parseInt(this.getAttribute('data-count'), 10);
        const isLiked = this.classList.contains('text-primary');

        if (isLiked) {
          this.classList.remove('text-primary');
          this.classList.add('text-on-surface-variant');
          icon.style.fontVariationSettings = "'FILL' 0";
          counter.textContent = currentCount + ' me gusta';
        } else {
          this.classList.add('text-primary');
          this.classList.remove('text-on-surface-variant');
          icon.style.fontVariationSettings = "'FILL' 1";
          counter.textContent = (currentCount + 1) + ' me gusta';
        }
      });
    }

    // Add new note creation micro-interaction
    const saveBtn = document.getElementById('save-note-btn');
    const reflectionInput = document.getElementById('reflection-input');
    const notesContainer = document.getElementById('notes-container');

    if (saveBtn && reflectionInput && notesContainer) {
      saveBtn.addEventListener('click', () => {
        const text = reflectionInput.value.trim();
        if (!text) {
          reflectionInput.focus();
          return;
        }

        const newArticle = document.createElement('article');
        newArticle.className = 'flex flex-col rounded-lg bg-surface-container p-space-md shadow-sm relative overflow-hidden transition-all hover:bg-surface-container-high animate-fadeIn';

    // Dismissal Handlers
    function closeDrawer() {
      if (!drawer || !scrim) return;
      drawer.style.transform = 'translateY(100%)';
      scrim.style.opacity = '0';
      scrim.style.pointerEvents = 'none';
      setTimeout(() => {
        drawer.style.display = 'none';

    if (closeBtn) closeBtn.addEventListener('click', closeDrawer);
    if (scrim) scrim.addEventListener('click', closeDrawer);

    // Subtle drag-to-dismiss hint
    let startY = 0;
    let currentY = 0;

    if (dragArea) {
      dragArea.addEventListener('touchstart', (e) => {
        startY = e.touches[0].clientY;
      }, { passive: true });

      dragArea.addEventListener('touchmove', (e) => {
        currentY = e.touches[0].clientY;
        const delta = currentY - startY;
        if (delta > 0) {
          drawer.style.transform = `translateY(${delta}px)`;
        }
      }, { passive: true });

      dragArea.addEventListener('touchend', () => {
        const delta = currentY - startY;
        if (delta > 120) {
          closeDrawer();
        } else {
          drawer.style.transform = 'translateY(0px)';

  // Interactive like toggle with delight feedback
  const likeButtons = document.querySelectorAll('.like-btn');
  const toast = document.getElementById('toast-pill');
  const toastText = document.getElementById('toast-text');
  let toastTimer;

  function showToast(msg) {
    if (toastTimer) clearTimeout(toastTimer);
    toastText.textContent = msg;
    toast.classList.remove('opacity-0', 'translate-y-2');
    toast.classList.add('opacity-100', 'translate-y-0');
    toastTimer = setTimeout(() => {
      toast.classList.remove('opacity-100', 'translate-y-0');
      toast.classList.add('opacity-0', 'translate-y-2');

  likeButtons.forEach(btn => {
    btn.addEventListener('click', (e) => {
      e.stopPropagation();
      const icon = btn.querySelector('.material-symbols-outlined');
      const isLiked = btn.getAttribute('data-liked') === 'true';

      if (isLiked) {
        btn.setAttribute('data-liked', 'false');
        icon.style.fontVariationSettings = "'FILL' 0";
        icon.textContent = 'favorite_border';
        btn.classList.remove('text-primary');
        btn.classList.add('text-on-surface-variant');
        showToast('Eliminado de versículos guardados');
      } else {
        btn.setAttribute('data-liked', 'true');
        icon.style.fontVariationSettings = "'FILL' 1";
        icon.textContent = 'favorite';
        btn.classList.remove('text-on-surface-variant');
        btn.classList.add('text-primary');
        showToast('Guardado en tus favoritos');
      }
    });
  });

  // Share action simulation
  const shareButtons = document.querySelectorAll('.share-btn');
  shareButtons.forEach(btn => {

      e.stopPropagation();
      if (navigator.share) {
        navigator.share({

          text: '«Jehová es mi pastor; nada me faltará.» — Salmos 23:1',
          url: window.location.href,

        showToast('Versículo copiado al portapapeles');
      }

  // PWA Install simulation
  const installBtn = document.getElementById('pwa-install-btn');
  if (installBtn) {

        installBtn.classList.add('bg-surface-container', 'text-on-surface-variant');
        showToast('Palabra Viva agregada a inicio');
      }, 1200);

      tab.classList.add('bg-primary', 'text-on-primary');
